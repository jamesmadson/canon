# Kits v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship Canon's kit product: four hand-curated kit pages, each offering a generated `install.sh` and a hand-authored digest, plus a kits index, nav entry, and homepage strip.

**Architecture:** A new `kits` content collection references skills by slug (integrity enforced by a corpus test). One pure function generates the install script from a kit's resolved skills; a static Astro endpoint ships it as a real file. Pages reuse the site's existing hairline-tile and monochrome type language. Kit editorial content (essays, digests) is authored by the controller/curator directly — Task 3 is explicitly curator-executed, not dispatched.

**Tech Stack:** Astro 5 (content collections + static endpoints) + TypeScript strict + Zod + Vitest + Tailwind v4.

## Global Constraints

- **Fully monochrome.** No color anywhere; only tokens `ink`, `ink-soft`, `rule`, `rule-strong`, `surface`, `bg`. Hover/focus = value shifts. Any accent color is a defect.
- No new npm dependencies.
- New interactive elements: ≥44px tap targets; nav links use the existing pattern `py-3 text-ink-soft transition-colors hover:text-ink focus-visible:text-ink`.
- Nav order after change: Kits, Gallery, Submit, Newsletter. Homepage kits section heading text is exactly `Kits`; index page h1 is exactly `Kits`.
- Generated script must be byte-deterministic (no timestamps) and log clone progress to **stderr** (stdout corrupts `$(clone …)` command substitution — bug found in the prototype).
- `digestPath` convention: `/kits/<slug>-digest.md`, file at `public/kits/<slug>-digest.md`.
- Kit lineups are spec-locked (see `docs/superpowers/specs/2026-08-11-kits-v1-design.md` §The four kits). Slugs: `full-redesign`, `marketing-site`, `product-ui`, `mobile-first-review`.
- Reference spec: `docs/superpowers/specs/2026-08-11-kits-v1-design.md`.

- [ ] **Step 3: Implement `src/lib/buildInstallScript.ts`**

```ts
import { parseSourceUrl } from './parseSourceUrl';

export interface KitSkillInput {
  slug: string;
  sourceUrl: string;
  hubPath: string;
  companionPaths: string[];
}

export interface KitScriptInput {
  slug: string;
  name: string;
  skills: KitSkillInput[];
}

function dirname(p: string): string {
  const i = p.lastIndexOf('/');
  return i === -1 ? '' : p.slice(0, i);
}

function joinPath(a: string, b: string): string {
  if (!a) return b;
  if (!b) return a;
  return `${a}/${b}`;
}

export function buildInstallScript(kit: KitScriptInput): string {
  const installLines: string[] = [];

  for (const skill of kit.skills) {
    const { owner, repo, path } = parseSourceUrl(skill.sourceUrl);
    const repoSpec = `${owner}/${repo}`;
    const installPath = joinPath(path, dirname(skill.hubPath));

    if (installPath === '') {
      installLines.push(`install_skill ${repoSpec} . ${skill.slug}`);
    } else {
      installLines.push(`install_skill ${repoSpec} ${installPath}`);
    }

    for (const companion of skill.companionPaths) {
      installLines.push(`install_skill ${repoSpec} ${companion}`);
    }
  }

  return `#!/usr/bin/env bash
#
# Canon kit: ${kit.name}
# Installs this kit's skills into ./.claude/skills/ of the current project.
#
# Usage: run from your project root:
#   bash install.sh
#
# Re-running updates everything in place (each skill folder is replaced).
# Generated by Canon (${kit.slug}) — https://canon.design

set -uo pipefail

DEST="$(pwd)/.claude/skills"
TMP="$(mktemp -d)"
FAILURES=0
trap 'rm -rf "$TMP"' EXIT

clone() { # clone <org/repo> once, shallow; prints dir on stdout, logs on stderr
  local repo="$1" dir="$TMP/\${1//\\//__}"
  if [ ! -d "$dir" ]; then
    echo "→ cloning $repo" >&2
    git clone -q --depth 1 "https://github.com/$repo" "$dir" >&2
  fi
  echo "$dir"
}

install_skill() { # install_skill <org/repo> <path-in-repo|.> [dest-name]
  local repo="$1" path="$2" name="\${3:-$(basename "$2")}"
  local src; src="$(clone "$repo")"
  if [ "$path" != "." ]; then src="$src/$path"; fi
  if [ ! -d "$src" ]; then
    echo "  !! $repo/$path not found — skipped" >&2
    FAILURES=$((FAILURES + 1))
    return 0
  fi
  rm -rf "$DEST/$name"
  cp -R "$src" "$DEST/$name"
  rm -rf "$DEST/$name/.git"
  echo "  ✓ $name  (from $repo)"
}

mkdir -p "$DEST"
echo "Installing the ${kit.name} kit into $DEST"
echo ""

${installLines.join('\n')}

echo ""
echo "Done. $(ls -1 "$DEST" | wc -l | tr -d ' ') skill folders now in .claude/skills/"
if [ "$FAILURES" -gt 0 ]; then
  echo "WARNING: $FAILURES skill(s) failed to install — see !! lines above." >&2
fi
echo ""
echo "Next: start a Claude Code session in this project — skills auto-load"
echo "from .claude/skills/ and trigger on matching work."
exit "$FAILURES"
`;
}
```

**Template-escaping note for the implementer:** the emitted bash uses `${…}` parameter expansions (`\${1//\\//__}`, `\${3:-…}`) which MUST be escaped inside the TypeScript template literal exactly as shown, while `${kit.name}`, `${kit.slug}`, and `${installLines.join('\n')}` are real TS interpolations. `$(basename "$2")`, `$(clone "$repo")`, `$(pwd)`, `$(mktemp -d)`, `$((FAILURES + 1))` and friends are NOT TS syntax (no braces) and pass through literally. After implementing, eyeball the emitted script from a test run (`console.log` once, or inspect the assertion diff) to confirm the bash reads correctly.

```ts
import matter from 'gray-matter';
import { kitSchema } from '../../src/content/kitSchema';

const kitFiles = readdirSync(kitsDir).filter((f) => f.endsWith('.mdx'));
const skillSlugs = new Set(
  readdirSync(skillsDir)
    .filter((f) => f.endsWith('.mdx'))
    .map((f) => skillSchema.parse(matter(readFileSync(path.join(skillsDir, f), 'utf-8')).data).slug)
);

describe('kits corpus integrity', () => {
  it('has exactly the four launch kits', () => {
    expect(kitFiles.length).toBe(4);
  });

  for (const file of kitFiles) {
    const { data } = matter(readFileSync(path.join(kitsDir, file), 'utf-8'));

    it(`${file} has valid frontmatter matching kitSchema`, () => {
      expect(() => kitSchema.parse(data)).not.toThrow();
    });

    it(`${file}'s skill references all resolve to real skills`, () => {
      const kit = kitSchema.parse(data);
      for (const phase of kit.phases) {
        for (const entry of phase.entries) {
          expect(skillSlugs.has(entry.skill), `${entry.skill} referenced by ${file} does not exist`).toBe(true);
        }
      }
    });

    it(`${file} references each skill at most once across phases`, () => {
      const kit = kitSchema.parse(data);
      const all = kit.phases.flatMap((p) => p.entries.map((e) => e.skill));
      expect(new Set(all).size).toBe(all.length);
    });

    it(`${file}'s digest file exists in public/`, () => {
      const kit = kitSchema.parse(data);
      expect(existsSync(path.join(publicDir, kit.digestPath)), `${kit.digestPath} missing`).toBe(true);
    });
  }

  it('has unique kit slugs', () => {
    const slugs = kitFiles.map((f) => matter(readFileSync(path.join(kitsDir, f), 'utf-8')).data.slug);
    expect(new Set(slugs).size).toBe(slugs.length);
```

```ts
import type { APIRoute, GetStaticPaths } from 'astro';
import { getCollection } from 'astro:content';
import { buildInstallScript, type KitSkillInput } from '../../../lib/buildInstallScript';

export const getStaticPaths: GetStaticPaths = async () => {
  const kits = await getCollection('kits', ({ data }) => data.status === 'active');
  return kits.map((kit) => ({ params: { slug: kit.data.slug }, props: { kit } }));
};

export const GET: APIRoute = async ({ props }) => {
  const { kit } = props as { kit: { data: import('../../../content/kitSchema').Kit } };
  const allSkills = await getCollection('skills');
  const bySlug = new Map(allSkills.map((s) => [s.data.slug, s]));

  const skills: KitSkillInput[] = [];
  for (const phase of kit.data.phases) {
    for (const entry of phase.entries) {
      const skill = bySlug.get(entry.skill);
      if (!skill) throw new Error(`Kit ${kit.data.slug} references unknown skill "${entry.skill}"`);
      const hub = skill.data.fileTree.find((f) => f.path.split('/').pop() === 'SKILL.md');
      if (!hub) throw new Error(`Skill ${entry.skill} has no SKILL.md in its fileTree`);
      skills.push({
        slug: skill.data.slug,
        sourceUrl: skill.data.sourceUrl,
        hubPath: hub.path,
        companionPaths: skill.data.companionPaths,
      });
    }
  }

  const script = buildInstallScript({ slug: kit.data.slug, name: kit.data.name, skills });
  return new Response(script, {
    headers: { 'Content-Type': 'text/x-shellscript; charset=utf-8' },
  });
};
```
